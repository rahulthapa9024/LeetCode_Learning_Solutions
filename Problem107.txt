class Solution {
public:
    bool canFinish(int n, vector<vector<int>>& prerequisites) {
        vector<int> adj[n];
        vector<int> inorder(n,0);
        vector<int> ans;

        for( auto x : prerequisites ){
            adj[x[0]].push_back(x[1]);
            inorder[x[1]]++;
        }

        queue<int> q;
        for(int i = 0; i < n; i++){
            if( inorder[i] == 0 ){
                q.push(i);
            }
        }

        while( !q.empty() ){
            auto t = q.front();
            ans.push_back(t);
            q.pop();

            for( auto x : adj[t] ){
                inorder[x]--;
                if( inorder[x] == 0 ){
                    q.push(x);
                }
            }
        }

        return ans.size() == n;
    }
};
