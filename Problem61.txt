// Jump Game II
// Medium

class Solution {
public:
    int jump(vector<int>& nums) {
        int far = 0, near = 0, jumps = 0;

        while( far < nums.size() - 1 ){
            int farthers  = 0;
            for(int i = near; i <= far; i++){
               farthers = max(farthers, i+ nums[i]);
            }

            near = far + 1;
            far = farthers;
            jumps++;
        }
        return jumps;
    }
};
