// Reverse Bits
// Easy

class Solution {
public:
    int reverseBits(int n) {
        string str = bitset<32>(n).to_string();

        string temp = "";
        for(int i = str.length() - 1; i >= 0; i--){
            temp += str[i];
        }

        int ans = stoi(temp, nullptr, 2);

        return ans;
    }
};
