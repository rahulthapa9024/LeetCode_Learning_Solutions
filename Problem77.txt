// Happy Number
// Eays

class Solution {
public:
    int getNextNumber(int n){
        int output = 0;
        while( n > 0 ){
            int digit = n % 10;
            output += digit * digit;
            n /= 10;
        }

        return output;
    }
    bool isHappy(int n) {
        unordered_set<int> visit;
        // loop until the number is not in the set
        while( visit.find(n) == visit.end() ){
            visit.insert(n);
            n = getNextNumber(n);
            if( n == 1 ) return true;
        }

        return false;
    }
};
