// Zigzag Conversion
// Medium 

class Solution {
public:
    string convert(string s, int numRows) {
        if( numRows == 1 ) return s; //base case 

        string result;

        for(int i = 0; i < numRows; i++){

            int index = i;
            int down = 2 * ( numRows - 1 - i );
            int up = 2  * i;
            bool flag = true;

            while( index < s.length() ){
                result += s[index];
                if( i == 0 ){ //first row 
                    index += down;
                }
                else if( i == numRows - 1){// last row
                    index += up;
                }
                else{
                    index += (flag ? down : up ); // middle rows 
                    flag = !flag;
                }
            }
            
        }

        return result;
    }
};
