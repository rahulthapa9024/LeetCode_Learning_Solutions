// Asteroid Collision
// Medium 


class Solution {
public:
    vector<int> asteroidCollision(vector<int>& asteroids) {
        int n = asteroids.size();
        int i = 0;
        stack<int> st;

        while( i < n ){

            if( asteroids[i] > 0 ){
                st.push(asteroids[i]);
            }

            else{
                int curr = asteroids[i];
                bool destroyed = false;

                while( !st.empty() && st.top() > 0 ){
                     if (abs(curr) > st.top()) {
                        st.pop();
                    }
                    else if( abs(curr) == st.top() ){
                        st.pop();
                        destroyed = true;
                        break;
                    }
                    else{
                        destroyed = true;
                        break;
                    }

                }
                if (!destroyed) {
                    st.push(curr);
                }
            }
            i++;
        }

        vector<int> ans(st.size(),0);

        for(int j = st.size() - 1  ; j >= 0; j--){
            ans[j] = st.top();
            st.pop();
        }
        return ans;

    }
};
