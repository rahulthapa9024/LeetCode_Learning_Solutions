// Smallest Number in Infinite
// Medium

class SmallestInfiniteSet {
public:
    int currentSmallest;
    priority_queue<int,vector<int>,greater<int>> pq;
    unordered_set<int> st;
    SmallestInfiniteSet() {
        currentSmallest = 1;
    }
    
    int popSmallest() {
      if( !pq.empty()){
          int x = pq.top();
          pq.pop();

          st.erase(x);

          return x;
      }
      return currentSmallest++;
    }
    
    void addBack(int num) {
        if( num < currentSmallest && st.find(num) == st.end()){
            pq.push(num);
            st.insert(num);
        }
    }
};
