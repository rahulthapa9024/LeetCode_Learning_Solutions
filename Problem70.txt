// Gas Station
// Medium

class Solution {
public:
    int canCompleteCircuit(vector<int>& gas, vector<int>& cost) {
        int totalGas = 0;
        int totalCost = 0;
        for(int i = 0; i < gas.size(); i++){
            totalCost += cost[i];
            totalGas += gas[i];
        }

        if( totalCost > totalGas ){
            return -1;
        }

        int startIdx = 0;
        int currentGas = 0;
        for(int i = 0; i < gas.size(); i++){
            currentGas += gas[i] - cost[i];

            if( currentGas < 0 ){
                startIdx = i + 1;
                currentGas = 0;
            }
        }

        return startIdx;
    }
};
