// Word Search
// Medium

class Solution {
public:
    vector<vector<int>> directions = {{1,0},{0,1},{-1,0},{0,-1}};

    bool solve(int idx,int i,int j,int m,int n,vector<vector<char>>& board, string word){
        if( idx >= word.length() ) return true;

        if( i < 0 || j < 0 || i >= m || j >= n || board[i][j] == '$' ) return false;
        
        if( board [i][j] != word[idx] ) return false;

        char temp = board[i][j];
        board[i][j] = '$';

        for( auto& dir : directions ){
            int new_i = i + dir[0];
            int new_j = j + dir[1];

            if( solve(idx+1,new_i,new_j,m,n,board,word) ){
                return true;
            }
        }

        board[i][j] = temp;

        return false;
    }

    bool exist(vector<vector<char>>& board, string word) {
        int m = board.size();
        int n = board[0].size();

        for(int i = 0; i < m; i++){
            for(int j = 0; j < n; j++){
                if( board[i][j] == word[0] && solve(0,i,j,m,n,board,word) ){
                    return true;
                }
            }
        }

        return false;
    }
};
