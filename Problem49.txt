// Successful Pairs of Spells and Potions
// Medium 

class Solution {
public:
    vector<int> successfulPairs(vector<int>& spells, vector<int>& potions, long long success) {
       int m = spells.size();
       int n = potions.size();

       sort(potions.begin(),potions.end());
       vector<int> ans;
        int maxPotionValue = potions[n-1];
       for( int i =0; i < m; i++){
        int spell = spells[i];

        long long minPortion = ceil((1.0*success)/spell);
            if(minPortion > maxPotionValue){
                ans.push_back(0);
                continue;
            }
        int index = lower_bound(begin(potions),end(potions),minPortion) - begin(potions);
        int count = n - index;

        ans.push_back(count);
       }

       return ans;
    }
};
