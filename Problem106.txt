// Evaluate Division
// Medium

class Solution {
public:
    void dfs( unordered_map<string,vector<pair<string,double>>>& adj,string src,string des,unordered_set<string> visited,double product,double& ans){
        if( visited.count(src) ){
            return;
        }

        visited.insert(src);

        if( src == des ){
            ans = product;
        }
        for( auto& p : adj[src] ){
            if( ans != -1.0 )return;
            string v = p.first;
            double val = p.second;

            dfs(adj,v,des,visited,product * val,ans);
        }
    }

    vector<double> calcEquation(vector<vector<string>>& equations, vector<double>& values, vector<vector<string>>& queries) {
        unordered_map<string,vector<pair<string,double>>> adj;
        int n = equations.size();

        // making of adj matrix
        for(int i = 0; i < n; i++){
            string u = equations[i][0];
            string v = equations[i][1];
            double val = values[i];

            adj[u].push_back({v, val});
            adj[v].push_back({u, 1.0 / val});
        }

        vector<double> result;
        for( auto& query : queries ){
            string src = query[0];
            string des = query[1];

            double ans = -1.0;

            if( adj.find(src) == adj.end() || adj.find(des) == adj.end()){
                result.push_back(-1.0);
                continue;
            }

            unordered_set<string> visited;
            dfs(adj,src,des,visited,1.0,ans);
            result.push_back(ans);
        }
        return result;
    }
};
