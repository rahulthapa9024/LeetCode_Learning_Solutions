// Dota2 Senate
// Medium

class Solution {
public:
    string predictPartyVictory(string senate) {
        queue<int> radinat;
        queue<int> dire;
        int n = senate.length();

        for( int i =0; i < n; i++ ){
            if( senate[i] == 'R'){
                radinat.push(i);
            }
            else{
                dire.push(i);
            }
        }

        while( !radinat.empty() && !dire.empty() ){
            int r = radinat.front();
            int d = dire.front();

            radinat.pop();
            dire.pop();

            if( r < d ){
                radinat.push( r + n );
            }
            else{
                dire.push( d + n );
            }
        }
        return radinat.empty() ? "Dire" : "Radiant";
    }
};
