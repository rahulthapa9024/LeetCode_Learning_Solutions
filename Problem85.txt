// Copy List with Random Pointer
// Medium

/*
// Definition for a Node.
class Node {
public:
    int val;
    Node* next;
    Node* random;
    
    Node(int _val) {
        val = _val;
        next = NULL;
        random = NULL;
    }
};
*/

class Solution {
public:
    Node* copyRandomList(Node* head) {
        if(!head) return nullptr;

        Node* cur = head;
        while(cur){
            Node* newNode = new Node(cur->val);
            newNode->next = cur->next;
            cur->next = newNode;
            cur = newNode->next;
        }

        cur = head;

        while(cur){
            if(cur->random != nullptr){
                cur->next->random = cur->random->next;
            }
            cur = cur->next->next;
        }

        cur = head;
        Node* newHead = head->next;
        Node* newCurr = newHead;

        while(cur){
            cur->next = newCurr->next;
            cur = cur->next;
            if( cur ){
                newCurr->next = cur->next;
                newCurr = newCurr->next;
            }
        }

        return newHead;

    }
};
