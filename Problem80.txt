// Minimum Number of Arrows to Burst Balloons
// Medium

class Solution {
public:
    int findMinArrowShots(vector<vector<int>>& points) {
        sort(points.begin(),points.end());
        int count = 1;
        vector<int> prev = points[0];

        for(int i = 1; i < points.size(); i++){
            vector<int> current = points[i];
            if( prev[1] < current[0] ){ // there is no overlap so we need another arrow and make the current balloon previous
                count++;
                prev = current;
            }
            else{ // there is an overlap so make the prev have the range where a balloon must overlap so we can vurst it with the same arrow
                prev[0] = max(prev[0],current[0]);
                prev[1] = min(prev[1],current[1]);
            }
        }

        return count;
    }
};
