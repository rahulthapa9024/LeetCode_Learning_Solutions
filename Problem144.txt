// Maximal Square
// Medium

class Solution {
public:
    int maximalSquare(vector<vector<char>>& matrix) {
        int m = matrix.size();
        int n = matrix[0].size();
        int t[301][301];
        memset(t,0,sizeof(t));
        int maxSqu = INT_MIN;

        for(int i = 0; i < m; i++){
            for(int j = 0; j < n; j++){
                if( matrix[i][j] == '1' ){
                    if( i == 0 || j == 0 ){
                        t[i][j] = 1;
                    }
                    else{
                    t[i][j] = 1 +
                    min({t[i-1][j],t[i][j-1],t[i-1][j-1]});
                }
                }
                maxSqu = max(maxSqu,t[i][j]);
            }
        }

        return maxSqu * maxSqu;
    }
};
