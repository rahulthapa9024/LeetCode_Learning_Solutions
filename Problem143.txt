// Interleaving String
// Medium

class Solution {
public:
    int m;
    int n;
    int N;
    int t[101][101];

    bool solve(string& s1, string& s2, string& s3,int i,int j){
        if( i == m && j == n && i+j == N ) return true;

        if( i+j >= N ) return t[i][j] = false;

        if( t[i][j] != -1 ) return t[i][j];

        bool result = false;

        if( s1[i] == s3[i+j] ){
            result = solve(s1,s2,s3,i+1,j);
        }

        if( result ) return result;

        if( s2[j] == s3[i+j] ){
            result = solve(s1,s2,s3,i,j+1);
        }

        return t[i][j] = result;
    }

    bool isInterleave(string s1, string s2, string s3) {
        m = s1.length();
        n = s2.length();
        N = s3.length();
        memset(t,-1,sizeof(t));

        return solve(s1,s2,s3,0,0);
    }
};
