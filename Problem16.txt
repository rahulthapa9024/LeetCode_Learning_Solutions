// Max Consecutive Ones III
// Medium

class Solution {
public:
    int longestOnes(vector<int>& nums, int k) {
        int l = 0, zeroes = 0, maxlen = 0;

        for(int r = 0; r < nums.size(); r++) {
            if(nums[r] == 0) zeroes++;

            while(zeroes > k) {
                if(nums[l] == 0) zeroes--;
                l++;
            }

            maxlen = max(maxlen, r - l + 1);
        }

        return maxlen;
    }
};
