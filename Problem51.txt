class Solution {
public:
    int FindMax(vector<int>& piles){
        return *max_element(piles.begin(),piles.end());
    }

    long long calculateTotalHours(vector<int>& piles, int speed){
        long long totalHour = 0;
        for( int bananas : piles){
            totalHour += (bananas + speed - 1 ) / speed;
        }
        return totalHour;
    }
    int minEatingSpeed(vector<int>& piles, int h) {
        int low = 1;
        int high = FindMax(piles);
        while( low <= high){
            int mid = low + (high - low) / 2;
            long long totalHour = calculateTotalHours(piles,mid);
            if( totalHour <= h){
                high = mid - 1;
            }
            else{
                low = mid + 1;
            }
        }
        return low;
    }
};
