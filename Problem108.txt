class Solution {
public:
    vector<int> findOrder(int n, vector<vector<int>>& prerequisites) {
        vector<int> adj[n];
        vector<int> inorder(n,0);
        vector<int> ans;

        for( auto x : prerequisites ){
            adj[x[1]].push_back(x[0]);
            inorder[x[0]]++;
        }

        queue<int> q;
        for(int i = 0; i < n; i++){
            if( inorder[i] == 0 ){
                q.push(i);
            }
        }

        while( !q.empty() ){
            int node = q.front();
            q.pop();
            ans.push_back(node);

            for( auto x : adj[node] ){
                inorder[x]--;
                if( inorder[x] == 0 ){
                    q.push(x);
                }
            }
        }

        if( ans.size() == n ) return ans;

        return {};
    }
};
